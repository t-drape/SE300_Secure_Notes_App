- generate_summary(text)
    sentences = split_sentences(text)

    if number of sentences == 0
        return error

    tokens = preprocess_text(text)

    if number of tokens == 0
        return error

    word_frequencies = compute_word_frequencies(tokens)

    sentence_scores = score_sentences(sentences)

    top_sentences = select_top_sentences(sentence_scores)

    sort top_sentences by original sentence order

    summary = join top_sentences into one text

    return summary


- get_top_keywords(text)
    tokens = preprocess_text(text)

    if number of tokens == 0
        return error

    keyword_frequencies = count_word_frequencies(tokens)

    keywords = select top keywords from keyword_frequencies

    return keywords
